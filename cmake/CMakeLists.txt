# Top-level CMake project for building adb with the Android NDK.
#
# fetch-sources.sh copies this file to <work>/vendor/CMakeLists.txt, next to
# the AOSP checkouts. The adb, libbase and libusb target definitions come
# from nmeum/android-tools (CMakeLists.{adb,libbase,libusb}.txt); this file
# provides everything they expect from their parent project.
cmake_minimum_required(VERSION 3.20)
project(adb_for_arm64 LANGUAGES C CXX)

if(NOT ANDROID)
	message(FATAL_ERROR "Configure with the Android NDK toolchain file "
		"(-DCMAKE_TOOLCHAIN_FILE=<ndk>/build/cmake/android.toolchain.cmake)")
endif()

include(GNUInstallDirs)

set(DEPS_PREFIX "" CACHE PATH "Prefix holding the static protobuf/brotli/lz4/zstd builds")
set(PLATFORM_TOOLS_VERSION "unknown" CACHE STRING "Version printed by adb --version")
set(ADB_BUILD_ID "adb-for-arm64" CACHE STRING "Build id printed by adb --version")

if(NOT DEPS_PREFIX)
	message(FATAL_ERROR "Set DEPS_PREFIX to the output of build-deps.sh")
endif()

# Options read by the nmeum/android-tools CMake lists.
set(ANDROID_TOOLS_USE_BUNDLED_LIBUSB ON)
set(ANDROID_TOOLS_LIBUSB_ENABLE_UDEV OFF)
set(ANDROID_TOOLS_ADB_ENABLE_MDNS OFF)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_EXTENSIONS ON)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_EXTENSIONS ON)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_compile_options(-Wno-attributes -ftrivial-auto-var-init=zero)
add_compile_definitions(_FILE_OFFSET_BITS=64 _LARGEFILE64_SOURCE)

# Headers normally generated by Soong.
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/platform_tools_version.h"
	"#define PLATFORM_TOOLS_VERSION \"${PLATFORM_TOOLS_VERSION}\"\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/build/version.h"
"#ifndef BUILD_VERSION_H
#define BUILD_VERSION_H
#include <string>
namespace android {
namespace build {
std::string GetBuildNumber() {
	return \"${ADB_BUILD_ID}\";
}
}  // namespace build
}  // namespace android
#endif
")
# libziparchive's zip_writer.h only needs FRIEND_TEST from googletest.
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/gtest/gtest_prod.h"
"#pragma once
#define FRIEND_TEST(test_case_name, test_name) friend class test_case_name##_##test_name##_Test
")

# Bundled AOSP libraries.
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory(boringssl EXCLUDE_FROM_ALL)
add_subdirectory(fmtlib EXCLUDE_FROM_ALL)

# Static libraries from build-deps.sh, exposed under the target names the
# nmeum/android-tools lists use for their pkg-config lookups.
function(import_static_lib target)
	string(MAKE_C_IDENTIFIER "${target}" var)
	find_library(${var}_PATH NAMES ${ARGN}
		PATHS "${DEPS_PREFIX}/lib" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH REQUIRED)
	add_library(${target} STATIC IMPORTED GLOBAL)
	set_target_properties(${target} PROPERTIES
		IMPORTED_LOCATION "${${var}_PATH}"
		INTERFACE_INCLUDE_DIRECTORIES "${DEPS_PREFIX}/include")
endfunction()

import_static_lib(PkgConfig::libbrotlicommon libbrotlicommon.a libbrotlicommon-static.a)
import_static_lib(PkgConfig::libbrotlidec libbrotlidec.a libbrotlidec-static.a)
import_static_lib(PkgConfig::libbrotlienc libbrotlienc.a libbrotlienc-static.a)
import_static_lib(PkgConfig::liblz4 liblz4.a)
import_static_lib(PkgConfig::libzstd libzstd.a)
set_property(TARGET PkgConfig::libbrotlidec PkgConfig::libbrotlienc
	APPEND PROPERTY INTERFACE_LINK_LIBRARIES PkgConfig::libbrotlicommon)

# Protobuf: pass Protobuf_PROTOC_EXECUTABLE (host), Protobuf_INCLUDE_DIR and
# Protobuf_LIBRARY (target) on the command line.
find_package(Protobuf REQUIRED)
set(THREADS_PREFER_PTHREAD_FLAG ON)
find_package(Threads REQUIRED)

include_directories(${Protobuf_INCLUDE_DIRS} "${DEPS_PREFIX}/include")
include_directories(${CMAKE_CURRENT_BINARY_DIR})

include(CMakeLists.libusb.txt)
include(CMakeLists.libbase.txt)

add_library(libzip STATIC
	libziparchive/zip_archive.cc
	libziparchive/zip_archive_stream_entry.cc
	libziparchive/zip_cd_entry_map.cc
	libziparchive/zip_error.cpp
	libziparchive/zip_writer.cc)
target_compile_definitions(libzip PRIVATE -DZLIB_CONST)
target_include_directories(libzip PUBLIC
	libziparchive/include
	libziparchive/incfs_support/include
	core/include
	logging/liblog/include
	libbase/include)

include(CMakeLists.adb.txt)

install(TARGETS adb RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
